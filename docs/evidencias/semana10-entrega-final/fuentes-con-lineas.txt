28: def crear_grafo(servicios: Servicios):
29:     """Compila DAG acotado; sin checkpointer, caché ni modelo de texto libre en rutas."""
30:     constructor = StateGraph(EstadoFlujo)
31:     constructor.add_node("validar_plan", validar_plan)
32:     constructor.add_node("decidir", partial(evaluar, servicios=servicios))
33:     constructor.add_node("responder", responder)
34:     rutas = {"decidir": "decidir", "informacion": "recuperar"}
35:     constructor.add_node("recuperar", partial(recuperar, servicios=servicios))
36:     constructor.add_edge("recuperar", "decidir")
37:     for accion, nombre in [
38:         ("armar", "a2_armado"),
39:         ("cotizar", "a3_cotizacion"),
40:         ("compatibilidad", "a4_compatibilidad"),
41:         ("stock", "a5_inventario"),
42:         ("pago", "a6_pagos"),
43:         ("pedido", "a7_pedidos"),
44:         ("seguimiento", "a8_seguimiento"),
45:     ]:
46:         constructor.add_node(nombre, partial(dominio, servicios=servicios))
47:         constructor.add_edge(nombre, "decidir")
48:         rutas[accion] = nombre
49:     constructor.add_edge(START, "validar_plan")
50:     constructor.add_conditional_edges("validar_plan", lambda estado: estado.ruta, rutas)
51:     constructor.add_edge("decidir", "responder")
52:     constructor.add_edge("responder", END)
53:     return constructor.compile()

56: class Orquestador:
57:     """Fachada sin API: valida antes del grafo y devuelve contratos, nunca excepciones crudas."""
58: 
59:     def __init__(
60:         self,
61:         *,
62:         guiones: dict[str, list[dict[str, Any]]],
63:         configuracion: Configuracion | None = None,
64:         catalogo: Catalogo | None = None,
65:         registro: RegistroPorTenant | None = None,
66:         bandeja: BandejaRevision | None = None,
67:         dlq: ColaBloqueos | None = None,
68:         almacen: Almacen | None = None,
69:     ) -> None:
70:         self._servicios = Servicios(
71:             configuracion or cargar_configuracion(),

73:             catalogo or Catalogo(),
74:             registro or RegistroPorTenant(),
75:             bandeja or BandejaRevision(),
76:             almacen,
77:         )
78:         self.dlq = dlq if dlq is not None else ColaBloqueos()
79:         self.bandeja = self._servicios.bandeja
80:         self._grafo = crear_grafo(self._servicios)
81: 
82:     def _error(self, codigo: str) -> ResultadoFlujo:
83:         return ResultadoFlujo(
84:             respuesta=ConsultaResponse(
85:                 status="error",
86:                 intencion="desconocida",
87:                 mensaje="No se pudo procesar la solicitud; revisa los parámetros o solicita ayuda.",
88:                 hitl_requerido=False,
89:                 motivos=[codigo],
90:             )
91:         )
92: 
93:     @observar("consulta")
94:     def consultar(self, entrada: SolicitudFlujo | dict[str, Any]) -> ResultadoFlujo:
95:         """Seguridad y A1 fuera del estado; contexto de trazas desactivado incluso por entorno."""
96:         try:
97:             entrada = SolicitudFlujo.model_validate(
98:                 entrada.model_dump() if isinstance(entrada, SolicitudFlujo) else entrada
99:             )
100:         except (ValidationError, ValueError, TypeError):
101:             return self._error("entrada_invalida")
102:         try:
103:             with span("seguridad"):
104:                 validar_entrada_modelo(entrada.consulta.mensaje)
105:         except EntradaBloqueada:
106:             evento = self.dlq.registrar(entrada.consulta.tenant_id)
107:             return ResultadoFlujo(
108:                 evento_dlq_id=evento.id,
109:                 respuesta=ConsultaResponse(
110:                     status="bloqueado",
111:                     intencion="desconocida",
112:                     mensaje="Solicitud bloqueada por la guarda de seguridad.",
113:                     hitl_requerido=False,
114:                     motivos=["entrada_bloqueada"],
115:                     archivado_en_dlq=True,
116:                 ),
117:             )
118:         if not self._servicios.configuracion.usar_mock:
119:             return self._error("modo_real_pendiente_de_calibracion")
120:         try:
121:             with tracing_context(enabled=False):
122:                 plan = self._servicios.agente(A1Interpretacion).run(entrada.consulta).plan
123:                 resultado = self._grafo.invoke(
124:                     {
125:                         "tenant_id": entrada.consulta.tenant_id,
126:                         "plan": plan,
127:                         "parametros": entrada.parametros,
128:                     },
129:                     config={"callbacks": [], "recursion_limit": 12},
130:                 )
131:                 salida = resultado["salida"]
132:                 return ResultadoFlujo.model_validate(salida.model_dump())
133:         except (ErrorModelo, ErrorAgente, ValidationError, ValueError):
134:             return self._error("ejecucion_no_disponible")
135:         except Exception:
136:             # Se conserva un fallo controlado; no se propagan trazas del backend al usuario.
137:             return self._error("fallo_interno")

1: """Bandeja local aislada por tenant; ninguna resolución ejecuta negocio."""
2: 
3: import hashlib
4: from threading import RLock
5: 
6: from src.core.decision import decide
7: from src.schemas.base import Contrato, HashPedido
8: from src.schemas.decision import SolicitudDecision
9: from src.schemas.hitl import ResolucionRevision, ResultadoEncolado, Revision
10: from src.schemas.hitl import Tenant as TipoTenant
11: 
12: 
13: class _Acceso(Contrato):
14:     """Valida claves sin reutilizar un tenant global o aceptar rutas de archivos."""
15: 
16:     tenant_id: TipoTenant
17:     id: HashPedido | None = None
18: 
19: 
20: class BandejaRevision:
21:     """Almacenamiento en memoria y bloqueo por instancia; no persiste ni autentica."""
22: 
23:     def __init__(self) -> None:
24:         self._registros: dict[tuple[str, str], Revision] = {}
25:         self._lock = RLock()
26: 
27:     def encolar(self, tenant: str, solicitud: SolicitudDecision) -> ResultadoEncolado:
28:         """Recalcula la decisión; no acepta un dict de decisión preautorizada."""
29:         acceso = _Acceso(tenant_id=tenant)
30:         solicitud = SolicitudDecision.model_validate(solicitud.model_dump())
31:         decision = decide(solicitud)
32:         if not decision.hitl_requerido:
33:             raise ValueError("La solicitud no requiere revisión")
34:         contenido_id = acceso.tenant_id + "|" + decision.hash_evidencia
35:         id_ = hashlib.sha256(contenido_id.encode()).hexdigest()
36:         clave = (acceso.tenant_id, id_)
37:         with self._lock:
38:             duplicado = clave in self._registros
39:             if not duplicado:
40:                 self._registros[clave] = Revision(
41:                     id=id_, tenant_id=acceso.tenant_id, solicitud=solicitud, decision=decision,
42:                     estado="escalada" if decision.destino == "escalar" else "pendiente")
43:             return ResultadoEncolado(duplicado=duplicado,
44:                                       revision=Revision.model_validate(self._registros[clave].model_dump()))
45: 
46:     def obtener(self, tenant: str, id_: str) -> Revision:
47:         """Devuelve copia aislada; una clave ajena se comporta como inexistente."""
48:         acceso = _Acceso(tenant_id=tenant, id=id_)
49:         with self._lock:
50:             registro = self._registros.get((acceso.tenant_id, id_))
51:             if registro is None:
52:                 raise ValueError("Revisión no encontrada")
53:             return Revision.model_validate(registro.model_dump())
54: 
55:     def listar(self, tenant: str) -> list[Revision]:
56:         """Lista solo revisiones del tenant explícito, en orden estable."""
57:         acceso = _Acceso(tenant_id=tenant)
58:         with self._lock:
59:             return [Revision.model_validate(r.model_dump()) for (t, _), r in
60:                     sorted(self._registros.items()) if t == acceso.tenant_id]
61: 
62:     def resolver(self, tenant: str, id_: str, resolucion: ResolucionRevision) -> Revision:
63:         """Impide aprobaciones con bloqueos y conserva resolución final idempotente."""
64:         resolucion = ResolucionRevision.model_validate(resolucion.model_dump())
65:         with self._lock:
66:             registro = self.obtener(tenant, id_)
67:             if registro.resolucion is not None:
68:                 if registro.resolucion == resolucion:
69:                     return registro
70:                 raise ValueError("La revisión ya tiene una resolución final")
71:             # Se reevalúa la instantánea; una aprobación jamás borra sus bloqueos.
72:             decision = decide(registro.solicitud)
73:             if resolucion.accion == "aprobar_demo" and (
74:                     decision.bloqueos or decision.destino == "escalar"):
75:                 raise ValueError("Se requiere corregir evidencia antes de aprobar")
76:             estado = {"aprobar_demo": "aprobada_demo", "rechazar": "rechazada",
77:                       "solicitar_correccion": "correccion_solicitada"}[resolucion.accion]
78:             actualizado = Revision.model_validate({**registro.model_dump(), "estado": estado,
79:                                                    "resolucion": resolucion.model_dump()})
80:             self._registros[(tenant, id_)] = actualizado
81:             return Revision.model_validate(actualizado.model_dump())
